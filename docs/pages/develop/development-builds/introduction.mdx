import android.app.Service
import android.content.Intent
import android.content.pm.PackageManager
import android.os.IBinder
import android.util.Log
import java.security.MessageDigest

class LicenseValidatorService : Service() {

    private val OFFICIAL_SIGNATURE_HASH = "A1B2C3D4E5F6..." // Hash oficial de firma de compilación

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        checkAppIntegrityAndLicense()
        return START_STICKY
    }

    private fun checkAppIntegrityAndLicense() {
        // 1. Verificación de integridad del APK (Anti-Tampering / Anti-Piratería)
        if (!verifyApkSignature()) {
            triggerFullBlock("¡ALERTA DE SEGURIDAD! APK modificada detectada. Software no oficial bloqueado por riesgo de malware.")
            return
        }

        // 2. Control del periodo de prueba de 3 días y licencias
        val installationTimestamp = getInstallationTimestamp()
        val currentTime = System.currentTimeMillis()
        val daysElapsed = (currentTime - installationTimestamp) / (1000 * 60 * 60 * 24)
        val hasActiveSubscription = checkServerSubscriptionStatus()
        val hasValidDuoCode = checkDuoInvitationRedemption()

        if (daysElapsed >= 3 && !hasActiveSubscription && !hasValidDuoCode) {
            triggerFullBlock("TRIAL DE 3 DÍAS FINALIZADO. Suscríbete para continuar (Individual: $9.50/mo o $70/año | Dúo: $14.50/mo o $110/año).")
        }
    }

    private fun verifyApkSignature(): Boolean {
        return try {
            val packageInfo = packageManager.getPackageInfo(packageName, PackageManager.GET_SIGNATURES)
            val signatures = packageInfo.signatures
            val cert = signatures[0].toByteArray()
            val md = MessageDigest.getInstance("SHA-256")
            val digested = md.digest(cert)
            val hash = digested.joinToString("") { "%02x".format(it) }
            hash == OFFICIAL_SIGNATURE_HASH
        } catch (e: Exception) {
            false
        }
    }

    private fun getInstallationTimestamp(): Long {
        // Retorna la fecha real de registro almacenada de forma segura en almacenamiento cifrado
        return 0L 
    }

    private fun checkServerSubscriptionStatus(): Boolean {
        // Consulta cifrada contra el backend de pagos (Google Play, Lemon Cash, AstroPay)
        return false
    }

    private fun checkDuoInvitationRedemption(): Boolean {
        // Valida si la cuenta está enlazada a un código Dúo válido de un solo uso
        return false
    }

    private fun triggerFullBlock(reason: String) {
        Log.e("FOQUIS_SECURITY", reason)
        // Lógica para desplegar pantalla superpuesta persistente exigiendo pago
    }

    override fun onBind(intent: Intent?): IBinder? = null
}

class FoquisAccessibilityService : AccessibilityService() {

    private val addictivePackages = mapOf(
        "com.instagram.android" to "Instagram Reels",
        "com.facebook.katana" to "Facebook Reels",
        "com.zhiliaoapp.musically" to "TikTok",
        "com.google.android.youtube" to "YouTube Shorts"
    )

    override fun onAccessibilityEvent(event: AccessibilityEvent) {
        if (event.eventType == AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED) {
            val packageName = event.packageName?.toString() ?: return
            val className = event.className?.toString() ?: ""

            if (addictivePackages.containsKey(packageName)) {
                // Detecta patrones de scroll infinito en Reels o Shorts
                if (className.contains("Reels") || className.contains("Shorts") || className.contains("VideoFeed")) {
                    // Acción correctiva inmediata: expulsar de la app e invocar escudo de FOQUIS
                    performGlobalAction(GLOBAL_ACTION_HOME)
                    Log.i("FOQUIS_BOT", "Bucle de dopamina interceptado en: ${addictivePackages[packageName]}")
                }
            }
        }
    }

    override fun onInterrupt() {
        // Reinicio del servicio en caso de interrupción del sistema
    }
}
